function isBlocked(raw){
  const n=norm(raw);
  // keep letters/digits; spaces separate words
  const words=n.replace(/[^\p{L}\p{N}]+/gu,' ').trim().split(' ').filter(Boolean);
  const compactAr=squash(words.join(''));
  const arOnly=squash(n.replace(/[^\u0600-\u06FF]/gu,''));
  const joined=words.join('');
  // letters spaced out like "ك ل ب": join single-letter runs
  const merged=[];let run='';
  for(const w of words){ if([...w].length===1){run+=w;} else {if(run){merged.push(run);run='';} merged.push(w);} }
  if(run)merged.push(run);
  const toks=[...new Set([...words,...merged])];
  // strong Arabic substring on squashed arabic-only text
  if(ARS.some(w=>w.length>=3&&arOnly.includes(w)))return true;
  // whole-word short Arabic
  for(const t of toks){const s=squash(t);
    for(const w of AR_WORD){const ws=squash(w);for(const p of AR_PREF){if(s===squash(p)+ws)return true;}}
    // suffix possessives
    for(const w of ['كس','زب','طيز','خول','نيك','عير']){const ws=squash(w);for(const p of AR_PREF)for(const suf of ['ك','ها','ي','كم','هم','ات','ين']) if(!(w==='خول'&&suf==='ات'))if(s===squash(p)+ws+suf)return true;}
  }
  // two-word phrases: "كس امك", "ابن الكلب"
  const j2=squash(words.join(''));
  if(/كس?(ا)?م?ك|كساخت|كسام|كس.?ام/.test('')){}
  if(/(^|\s)(و|ي|يا)?كس\s*(ام|ا?م|اخت|اب|دين)/.test(n.replace(/[^\p{L}\p{N}]+/gu,' ')))return true;
  if(/دين\s*(ام|اب)/.test(n.replace(/[^\p{L}\p{N}]+/gu,' ')))return true;
  if(/ابن\s*(ال)?(كلب|وسخ|متناك|شرموط|قحب|حمار|زنا)|اب?ن\s*ال?(كلب|متناك)|ولاد\s*(ال)?(كلب|متناك|شرموط|قحب)|بنت\s*(ال)?(كلب|وسخ|متناك|شرموط|قحب)/.test(n.replace(/[^\p{L}\p{N}]+/gu,' ')))return true;
  // Latin / Franco
  let l=n.replace(/[\u0600-\u06FF]/g,' ');
  const lw=l.replace(/[^a-z0-9$@!]+/g,' ').trim().split(' ').filter(Boolean).map(w=>w.replace(/[$]/g,'s').replace(/@/g,'a').replace(/!/g,'i'));
  const lm=[];let lr='';for(const w of lw){if(w.length===1)lr+=w;else{if(lr){lm.push(lr);lr='';}lm.push(w);}}if(lr)lm.push(lr);lw.push(...lm);const lj=squash(lw.join(''));
  const lj2=squash(lw.join('').replace(/0/g,'o').replace(/1/g,'i'));

  return false;
}
